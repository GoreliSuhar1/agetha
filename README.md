# instalation guide
# 1
`sudo yay -S agetha`
# 2
`mkdir -p ~/.config/agetha && echo '{"api_key": "YOUR GROQ API KEY HERE!!!"}' > ~/.config/agetha/config.json`
# How to start
`agetha`
# how to send massage
right click on window with Agetha and write a massage and press enter
